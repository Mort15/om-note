----------- UserParameter -----------
vim /etc/zabbix/zabbix_agent.conf
inlucde=./zabbix_agentd.d/*.conf

vim ./zabbix_agentd.d/os.conf
UserParameter=login_user,w|awk 'NR=1{print $5}'
定义login_user变量，赋值$(w|awk 'NR=1{print $5}')
加上之后重启zabbix，并且列出内部值
zabbix_agentd -t | grep login_user

----- 远程执行命令 -----
动作里可以死掉后自动重启
改zabbix权限 /etc/sudoers
zabbix ALL=(ALL) NOPASSWD: ALL

su - zabbix -s /bin/bash
systemctl restart nginx

----- script -----
Hostname=nginx.[*] 搭配脚本:
#! /bin/sh
case $1 in
active)
curl -s 127.0.0.1/nginx_status | awk ' NR == 1{print $NF}'

accepts)
curl -s 127.0.0.1/nginx_status | awk 'NR == 3{print $1}'
;;
handled)
curl -s 127.0.0.1/nginx_status | awk ' NR == 3{print $2}'
;;
requests)
curl -s 127.0.0.1/nginx_status | awk ' NR == 3{print $3}'

..*

echo "Usage: $0 [active | accepts | handled | requests]"

esac

----- SNMP -----
简单网络管理协议（SNMP）
监控网络设备: 路由 交换机
snmpv1不支持加密
snmpv2有加密
snmpv3增强加密，太慢

yum -y install net-snmp
/etc/snmp/snmpd.conf
snmpwalk -v2c -c public <IP> <OID>
