根据我们的毕生所学 sed [] -z expr 判断输入参数
#!/bin/bash
read -p "输入你的姓名: " name
name_test=$(echo $name | sed -r 's#[^0-9]##g' | wc -L)
[ $name_test -ne 0 ] && echo "你的名字里怎么还有编号?" && exit
[ -z $name ] && echo "你没有名字?" && exit
read -p "输入你的年龄: " age
expr $age + 1 &>/dev/null
[ $? -ne 0 ] && echo "请用阿拉伯数字。" && exit
echo "name: ${name}"
echo "age: ${age}"
