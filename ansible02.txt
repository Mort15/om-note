----------ansible变量------------
- hosts: web01
  vars: p1: wget
  tasks:
  - name: yum test
    yum:
      name: "{{ p1 }}"
      state: present

- hosts: web02
  vars:
    - p2: wget
    - p3: lrzsz
  tasks:
  - name: yum test
    yum:
      name:
        - "{{ p1 }}"
        - "{{ p2 }}"
      state: present

- hosts: web03
  vars:
    pack:
      - wget
      - lrzsz
  tasks:
  - name: yum test
    yum:
      #变量单独处理，就是彩色的
      name: "{{ pack }}"
      state: absent
----------ansible变量列表方式------------
- hosts: web01
  vars:
    - ip: 10.0.0.7
    - ho: web01
  tasks:
    - name: yum test
      yum:
        name: /root/{{ ho }}_{{ ip }}
        state: directory
#注意路劲必须是白色的，不能有彩色，不能加双引号
#单独出现变量，必须加双引号
----------ansible调用文件------------
- hosts: web01
  vars_files: ./v1.yml
  tasks:
    - name: yum test
      yum:
        name:
          - "{{ p1 }}"
          - "{{ p2 }}"
          - "{{ p3 }}"
          state: present
----------ansible调用多个文件------------
- hosts: web01
  vars_files:
    - v1.yml
    - ./v2.yml
    - v3.yml
  tasks:
    - name: yum test
      yum:
        name:
          - "{{ p1 }}"
          - "{{ p2 }}"
          - "{{ p3 }}"
          state: absent
#注意，这里的v1 v2 v3都不能是列表，列表不能再放进列表里。

----------ansible过滤使用内置变量------------
ansible web01 -m setup #通过这个，获取全部参数
ansible web01 -m setup -a filter="ipv4"
ansible web01 -m setup -a filter="default_ipv4_address"
ansible web01 -m setup -a filter="ansible_eth0_address"
ansible web01 -m setup -a filter="ansible_all_ipv4_address"
ansible web01 -m setup -a filter="ansible_fqdn"
ansible web01 -m setup -a filter="ansible_hostname"
ansible web01 -m setup|grep ansible_default_ipv4 -A 5 #向下取五行

----------ansible变量注册------------
- hosts: web01
  tasks:
    - name: test
      command: 'nginx -t'
      register: result_nginx  #定义时不用花括号

    - name: print result_nginx
      debug:
        msg: "{{ result_nginx.rc }}"  #输出rc就是反馈码
        msg: "{{ result_nginx.stderr_lines }}"  #stderr_lines是反馈结果


- hosts: webs
  vars_files:
    - a.yml  #调用变量
  tasks:
    - name: yum test
      yum:
        name: "{{ lamp.framework.php_package }}"
        state: absent
