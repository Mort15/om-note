--------CPU亲和affinity--------

解决性能损耗: 核心数量和Nginx进程不一致时，会不断切换，切换就会导致CPU缓存miss。
#防止有人不干活，有人干的太多。
#官方推荐和CPU核心数保持一致
lscpu
#查看 nginx worker 绑定对应 cpu
ps -eo pid,args,psr|grep [n]ginx

#自动就行了  affinity亲和力
worker_processes auto;
worker_cpu_affinity auto; #必须设置


worker_connections 10240; #大了没事，不能太小
worker_rlimit_nofile 35535; #每个work能打开的文件描述符上限，不能太小

server_tokens off; #隐藏nginx版本
client_max_body_size 200m; #上传大小限制

#文件高效传输
sendfile on;
tcp_nopush on;
#实时传输
tcp_nodelay on;
keepalived_timeout 65;

#Gzip 压缩
gzip on;
gzip_disable "MSIE [1-6]\.";
gzip_http_version 1.1;
gzip_comp_level 2;
gzip_buffers 16 8k;
gzip_min_length 1024;
gzip_types text/plain text/css application/json application/x-javascript text/xml application/xml
application/xml+rss text/javascript image/jpeg;
