#限制SSH连接频率
iptables -I INPUT -p tcp --dport 22 -m state --state NEW -m recent --set
iptables -I INPUT -p tcp --dport 22 -m state --state NEW -m recent --update --seconds 60 --hitcount 3 -j DROP
1, -m recent --set      记录set  来源的小本本
2, -m recent --update   找update  小本本里有没有记录过
3, --seconds 60 --hitcount 3 -j DROP 设定条件: "记考勤，60秒内旷工3次的直接开除。"

#NAT上网
echo 1 > /proc/sys/net/ipv4/ip_forward
iptables -t nat -A POSTROUTING -s 192.168.1.0/24 -o eth0 -j MASQUERADE
iptables -t nat -A POSTROUTING -s 192.168.1.0/24 -o eth0 -j SNAT --to-source 12.34.56.78
1, 公网ip不固定时: MASQUERADE   #因为要查路由表，速度会慢一点
2, 公网ip固定时: -j SNAT --to-source 12.34.56.78

#端口转发
iptables -t nat -A PREROUTING -p tcp --dport 9090 -j DNAT --to-destination 192.168.1.100:80
iptables -A FORWARD -p tcp -d 192.168.1.100 --dport 80 -j ACCEPT
# 篡改快递单上的收货地址-j DNAT --to-destination 192.168.1.100:80 

改变目标 PREROUTING DNAT 公网冒充内网IP进来，端口映射，
改变来源 POSTROUTING SNAT 内网冒充公网IP出去，上网刷视频，是家常便饭


#允许已建立连接
iptables -I INPUT -m state --state ESTABLISHED,RELATED -j ACCEPT

#日志记录 可以记录原本被DROP的包
iptables -I INPUT -j LOG --log-prefix "FW-DROP: " --log-level 4

可选项: -j LOG

第四模块_NAT网关与端口映射:
    DNAT  跳板机    隐藏自己
    SNAT  网关路由  上网刷视频
  题xx: DNAT  
    需求: "做公网跳板机保护自己"
    # 开启转发
    echo 1 > /proc/sys/net/ipv4/ip_forward
    # DNAT：将进来的流量转发给下一跳
    iptables -t nat -A PREROUTING -i eth0 -j DNAT --to-destination 下一跳IP
    # SNAT：将回包伪装成跳板自身
    iptables -t nat -A POSTROUTING -o eth1 -j MASQUERADE
    # 放行转发
    iptables -A FORWARD -j ACCEPT

  题xx: SNAT  
    #开启转发
    echo "net.ipv4.ip_forward = 1" >> /etc/sysctl.conf
    sysctl -p

    #策略路由：来源网段走指定出口
    ip rule add from 192.168.1.0/24 table 100
    ip route add default via 192.168.1.1 dev eth1 table 100

    #标记流量+策略路由
    iptables -t mangle -A PREROUTING -s 10.0.0.0/24 -j MARK --set-mark 1
    ip rule add fwmark 1 table 200
    ip route add default via 10.0.0.1 dev eth2 table 200

    #多线SNAT
    iptables -t nat -A POSTROUTING -s 192.168.1.0/24 -o eth0 -j MASQUERADE
    iptables -t nat -A POSTROUTING -s 192.168.2.0/24 -o eth1 -j MASQUERADE

    #查看路由
    ip route show
    ip rule list

    #保存配置
    service iptables save
    ip route save > /etc/iproute2/route.conf
