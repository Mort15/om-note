#!/bin/bash
os_ver=$(awk '{print $4}' /etc/redhat-release)
os_release=$(cat /etc/redhat-release)
backup_yum='mv /etc/yum.repos.d/Centos-Base.repo /etc/yum.repos.d/Centos-Base.repo.backup'
which wget &>/dev/null
if [ $? -ne 0 ];then
  echo "install...."
  yum -y install wget &>/dev/null
  if [ $? -eq 0 ];then
    echo "ok"
  else [ $? -eq 0 ]
    echo "no!!  install wget no"
    exit
  fi

fi

echo $os_release|grep CentOS
if [[ $? ]];then
  [ -e /etc/yum/repos.d/Centos-Base.repo] && $backup_yum
  [ -e /etc/yum.repos.d/ ] && wget -O /etc/yum.repos.d/Centos-Base.repo https://mirrors.aliyun.com/repo/Centos-${os_ver%%.*}.repo
fi
