nginx静态文件下载站点:
autoindex on; #启动下载站
alias #永远绝对路径
location #在root的基础上，子地址
root #指定文件地址，父地址
charset utf-8,gbk; #万国码和国标，解决中文乱码
autoindex_localtime on; #使用本地时间
autoindex_exact_size off; #文件大小
