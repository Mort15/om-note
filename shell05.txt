#1,不重id

read -p "你的名字: " name
[ -z $name ] && continue
[ ! -e tmp_shell_p2 ] && touch tmp_shell_p2
id=$(($RANDOM%100+1))
if [ $name = ok ];then
  exit
fi
if [ $(awk '{print $1}' tmp_shell_p2|grep -w $id) ];then
  continue
fi

echo $id $name

  echo "已经满员了"
  exit
fi
