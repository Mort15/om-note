#把新nginx替换旧的nginx
mv /usr/local/nginx/sbin/nginx /usr/local/nginx/sbin/nginxold
cp /nginx/core/nginx-1.16.1/objs/nginx /usr/local/nginx/sbin/nginx

#生成nginx.pid.oldbing
kill -USR2 $(cat /usr/local/logs/nginx.pid)
#退掉oldbin
kill -QUIT $(cat /usr/local/logs/nginx.pid.oldbin)
#这时候就完成版本升级了

nginx更好的升级:
mv ./nginx ./nginxold
#假如已经编译好./objs/nginx
#在nginx-1.16.1目录，在objs父目录
make upgrade
#可以看到make会自动发送 USR2 和 QUIT

#nginx改成系统服务方式
vim /usr/lib/systemd/system/nginx.service
[Unit]
#略，描述和文档。
After=network.target #表示当前服务在指定的服务之后启动

[Service]
Type=forking #最常用的类型，似乎是"长时间执行"都用这个。
PIDFile #指定pid路径
ExecStartPre #启动之前执行命令
ExecStart #systemctl start 命令
ExecReload #systemctl restart 命令
ExecStop #systemctl stop 命令
#
#服务会在启动时创建一个独立的临时目录,服务停止时，该目录会自动删除
PrivateTmp=true 

[Install]
#
#服务将在系统启动到 default.target 所指向的目标时启动
wantedBy=default.target

chmod 755 /usr/lib/systemd/system/nginx.service

#差异environment;bashrc;profile;
文件路径               | 生效范围   | 是否支持$扩展 | 加载时机
----------------------|-----------|-------------|---------------------------
~/.bashrc             | 当前用户   | 是          | 交互式非登录终端
~/.profile            | 当前用户   | 是          | 登录Shell（兼容性最好）
~/.bash_profile       | 当前用户   | 是          | 登录Shell（优先于.profile）
~/.bash_login         | 当前用户   | 是          | 登录Shell（优先于.profile，次于.bash_profile）
/etc/profile          | 所有用户   | 是          | 系统全局登录Shell
/etc/bash.bashrc      | 所有用户   | 是          | 系统全局Bash交互式终端
/etc/profile.d/*.sh   | 所有用户   | 是          | 登录Shell（推荐拆分脚本）
/etc/environment      | 所有用户   | 否          | 系统启动/登录（语法严格）
~/.pam_environment    | 当前用户   | 否          | PAM登录（极少用，语法严格）

server_name 和 listen 都可以区分不同网站
listen 127.0.0.1:8000 #不同ip;不同端口;
server_name _; #_表示所有;主要写域名;

server_name 星号不能出现在中间;
*.aliyun.com和aliyun.*是对的;

#\w 表示匹配任意字母、数字或下划线的字符
server_name ~^www\.(\w+)\.com;
$1 引用第一个括号匹配的内容
return 200 $1 $2;

#这样可以输出文本
return 200 'hahahaha';

listen 80 default_server 和 listen 80 的区别
第一个被加载的 server 块，是default_server。
或者用default_server手动指定默认。

加default_type text/plain;，不继承上层块。
如果没有可以继承的default_type，那么默认就是text/plain

为什么是text/plain？
text是大类
plain是text当中没有任何元数据的纯文本。
plain还有一层隐含语义:“这份内容是给人眼直接阅读的”。
plain老老实实把字符打印出来，千万别解析
application/octet-stream 别打开，直接让用户下载保存

HTTP页面缓存相关头信息，服务器判断发送新旧。
Expires	日期，会影响 Cache-Control
Cache-Control	配置信息
Last-Modified	最后修改时间
ETag	不懂，反正MD5值属于这里，好像是索引文件用的。

location ~ .*\.(html|js|css|png)$ {
	expires 1000; #缓存1000秒
	expires -1000; #(影响Cache-Control)配置过期缓存，会对比本地和服务器是否有变化。
	expires max; #十年
	add_header Cache-Control no-store; #主动设置，不缓存，每次都是新数据，有很多选项
}
