-------when判断-------
场景，当一个公司有centos有ubuntu。
获取ansible内部变量，通过grep找。
ansible web01 -m setup | grep -i centos
用when来判断CentOS系统
- hosts: webs
  tasks:
    - name: when
      yum:
        name: wget
        state: present
      when: ansible_distribution == "CentOS"
其他主机会有蓝色字体的跳过提示。
when: ansible_hostname == "web01"
when: ansible_distribution != "CentOS"
when: ansible_hostname is match "web01"
when: ansible_hostname is not match "web01"
when: ansible_hostname is search "web"
when: (ansible_hostname is match "web01") and (ansible_default_ipv4.address == "10.0.0.7")
when: (ansible_hostname is match "web01") or (ansible_default_ipv4['address'])

- hosts: web01
  tasks:
#定义变量，同时nginx -t检测配置文件
    - name: when
      command: nginx -t
      register: re_ng
#忽略报错向下执行
      ignore_errors: yes

#打印大量内容，从这里判断rc，可以后面删掉
    - name:
      debug:
        msg: "{{ re_ng }}"

#多个when写法
    - name: restarted nginx server
      systemd:
        name: nginx
        state: restarted
      when: re_ng.rc == "0"
      when: re_ng.rc is search "0"

-------loop-------
- hosts: web01
  tasks:
    - name: Start Nginx PHP Server
      systemd:
        name: "{{ item }}"
        state: started
      loop:
        - nginx
        - php-fpm

- hosts: web02
  tasks:
    - name: Start Nginx PHP Server
      file:
        name: /root/{{ item.nameee }}
        owner: "{{ item.ownerr }}"
        group: "{{ item.groupppp }}"
        mode: "{{ item.mode }}"
        state: touch
      loop:
        - { nameee: 1.txt, ownerr: root, groupppp: root, mode: "0777" }
        - { nameee: 2.txt, ownerr: root, groupppp: www, mode: "0600" }

-------loop创建用户实例-------
#只要一一对应的修改就行了
test01 uid 777
test02 uid 888

- hosts: web01
  tasks:
    - name: Create test User
      user:
        name: "{{ item.name }}"
        group: "{{ item.group }}"
        uid: "{{ item.uid }}"
        shell: "{{ item.shell }}"
        create_home: "{{ item.home }}"
        state: present
      loop:
        - { name: test01, uid: 777 , shell: /bin/bash, home: true}
        - { name: test02, uid: 888 , shell: /sbin/nologin, home: false}
